import * as ecs from "@8thwall/ecs";
import { gameData } from "../core/game-data";
import { OBJECT_PLACED_EVENT } from "./placement-system";
import { collectible, CollectibleType } from "../components/collectible";

// Fisher-Yates shuffle
function shuffle<T>(array: T[]): T[] {
  for (let i = array.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));

    [array[i], array[j]] = [array[j], array[i]];
  }

  return array;
}

ecs.registerComponent({
  name: "collectible-spawn-system",

  schema: {
    collectibleSpawnContainer: ecs.eid,

    burritoPrefab: ecs.eid,
    steakPrefab: ecs.eid,
    friesPrefab: ecs.eid,
    chiliPrefab: ecs.eid,
  },

  stateMachine: ({ world, eid, schemaAttribute, defineState }) => {
    defineState("waiting")
      .initial()

      .onEnter(() => {
        console.log("[Collectibles] Component initialized");

        world.events.addListener(
          world.events.globalId,
          OBJECT_PLACED_EVENT,

          () => {
            console.log(
              "[Collectibles] DriveZone placed. Spawning collectibles...",
            );

            spawnCollectibles(world, schemaAttribute.get(eid));
          },
        );
      });
  },
});

function spawnCollectibles(
  world: ecs.World,

  schema: {
    collectibleSpawnContainer: ecs.Eid;

    burritoPrefab: ecs.Eid;
    steakPrefab: ecs.Eid;
    friesPrefab: ecs.Eid;
    chiliPrefab: ecs.Eid;
  },
) {
  if (!gameData.driveZoneEid) {
    console.error("[Collectibles] No DriveZone instance.");

    return;
  }

  if (gameData.collectiblesSpawned) {
    console.warn("[Collectibles] Already spawned.");

    return;
  }

  // Get runtime spawn container

  const runtimeContainer = world.getInstanceEntity(
    gameData.driveZoneEid,

    schema.collectibleSpawnContainer,
  );

  console.log("[Collectibles] Runtime Container:", runtimeContainer);

  // Get all 24 points and shuffle

  const spawnPoints = shuffle([...world.getChildren(runtimeContainer)]);

  console.log("[Collectibles] Available points:", spawnPoints.length);

  // Select only 4

  const selectedPoints = spawnPoints.slice(0, 4);

  console.log("[Collectibles] Selected points:", selectedPoints);

  // Randomize food types

  const items = shuffle([
    {
      prefab: schema.burritoPrefab,
      type: CollectibleType.BURRITO,
      value: 1,
    },

    {
      prefab: schema.steakPrefab,
      type: CollectibleType.STEAK,
      value: 2,
    },

    {
      prefab: schema.friesPrefab,
      type: CollectibleType.FRIES,
      value: 3,
    },

    {
      prefab: schema.chiliPrefab,
      type: CollectibleType.CHILI,
      value: 4,
    },
  ]);

  selectedPoints.forEach((point, index) => {
    const item = items[index];

    if (!item) return;

    // Create food entity

    const collectibleEid = world.createEntity(item.prefab);

    // Attach collectible ECS component

    world.getEntity(collectibleEid).set(collectible, {
      type: item.type,
      value: item.value,
    });

    // Move to spawn point

    const position = world.transform.getWorldPosition(point);

    world.transform.setWorldPosition(
      collectibleEid,

      {
        x: position.x,
        y: position.y + 0.05,
        z: position.z,
      },
    );

    console.log(
      "[Collectibles] Spawned",
      collectibleEid,
      item.type,
      "at",
      position,
    );
  });

  gameData.collectiblesSpawned = true;

  console.log("[Collectibles] Spawn complete.");
}
