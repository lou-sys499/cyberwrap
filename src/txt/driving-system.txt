import * as ecs from "@8thwall/ecs";

import { gameData } from "../core/game-data";
import { GameState } from "../core/game-state";

ecs.registerComponent({
  name: "driving",

  schema: {
    speed: ecs.f32,
    turnSpeed: ecs.f32,
  },

  schemaDefaults: {
    speed: 0.05,
    turnSpeed: 0.05,
  },

  tick: (world, component) => {
    if (!gameData.truckEid) return;

    if (gameData.state !== GameState.DRIVING) return;

    const { eid } = component;

    const { speed, turnSpeed } = component.schema;

    // steering
    const rotation = ecs.math.quat.yRadians(
      -gameData.input.steering * turnSpeed,
    );

    world.transform.rotateSelf(eid, rotation);

    // throttle
    world.transform.translateSelf(eid, {
      z: gameData.input.throttle * speed,
    });
  },
});
