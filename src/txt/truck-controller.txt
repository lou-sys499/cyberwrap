import * as ecs from "@8thwall/ecs";
import { gameData } from "../core/game-data";

const MOVE_SPEED = 1.5;
const TURN_SPEED = 120;

ecs.registerComponent({
  name: "truck-controller",

  tick: (world, cursor) => {
    if (!gameData.canDrive) return;

    const dt = world.time.delta / 1000;

    const steering = gameData.input.steering;
    const throttle = gameData.input.throttle;

    const isMoving = Math.abs(throttle) > 0.05;

    // Only steer while moving
    if (isMoving && Math.abs(steering) > 0.05) {
      world.transform.rotateLocal(
        cursor.eid,
        ecs.math.quat.yDegrees(-steering * TURN_SPEED * dt),
      );
    }

    // Move forward/backward while a button is held
    if (isMoving) {
      world.transform.translateLocal(cursor.eid, {
        z: -throttle * MOVE_SPEED * dt,
      });
    }
  },
});
