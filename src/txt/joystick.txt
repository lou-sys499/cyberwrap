import * as ecs from "@8thwall/ecs";
import { gameData } from "../core/game-data";

ecs.registerComponent({
  name: "joystick",

  stateMachine: ({ defineState }) => {
    defineState("initial")
      .initial()
      .onEnter(() => {
        // ---------------------------------------------------------------------
        // Create joystick UI
        // ---------------------------------------------------------------------

        const SIZE = 100;
        const RADIUS = SIZE / 2;
        const THUMB_SIZE = 50;
        const MAX_DISTANCE = 25;

        const ui = document.createElement("div");
        ui.style.cssText = `
          position: fixed;
          left: 40px;
          bottom: 40px;
          width: ${SIZE}px;
          height: ${SIZE}px;
          border-radius: 50%;
          background: rgba(255,255,255,0.25);
          border: 2px solid rgba(255,255,255,0.45);
          touch-action: none;
          user-select: none;
          z-index: 9999;
        `;

        const thumb = document.createElement("div");
        thumb.style.cssText = `
          position: absolute;
          left: ${(SIZE - THUMB_SIZE) / 2}px;
          top: ${(SIZE - THUMB_SIZE) / 2}px;
          width: ${THUMB_SIZE}px;
          height: ${THUMB_SIZE}px;
          border-radius: 50%;
          background: white;
          pointer-events: none;
          transition: transform 0.02s linear;
        `;

        ui.appendChild(thumb);
        document.body.appendChild(ui);

        let dragging = false;

        function updateJoystick(clientX: number, clientY: number) {
          const rect = ui.getBoundingClientRect();

          const centerX = rect.left + rect.width / 2;
          const centerY = rect.top + rect.height / 2;

          let dx = clientX - centerX;
          let dy = clientY - centerY;

          const distance = Math.sqrt(dx * dx + dy * dy);

          // Clamp movement inside joystick radius
          if (distance > RADIUS) {
            dx = (dx / distance) * RADIUS;
            dy = (dy / distance) * RADIUS;
          }

          // Normalize to [-1, 1]
          const steering = dx / RADIUS;
          const throttle = -dy / RADIUS;

          gameData.input.steering = steering;
          gameData.input.throttle = throttle;

          // Visual thumb movement
          thumb.style.transform = `translate(${steering * MAX_DISTANCE}px, ${-throttle * MAX_DISTANCE}px)`;
        }

        function resetJoystick() {
          dragging = false;

          gameData.input.steering = 0;
          gameData.input.throttle = 0;

          thumb.style.transform = "translate(0px, 0px)";
        }

        // ---------------------------------------------------------------------
        // Pointer Events
        // Works for:
        //   ✔ Mouse
        //   ✔ Touch
        //   ✔ Stylus
        // ---------------------------------------------------------------------

        ui.addEventListener("pointerdown", (e: PointerEvent) => {
          dragging = true;
          ui.setPointerCapture(e.pointerId);
          updateJoystick(e.clientX, e.clientY);
        });

        ui.addEventListener("pointermove", (e: PointerEvent) => {
          if (!dragging) return;

          e.preventDefault();
          updateJoystick(e.clientX, e.clientY);
        });

        ui.addEventListener("pointerup", () => {
          resetJoystick();
        });

        ui.addEventListener("pointercancel", () => {
          resetJoystick();
        });

        ui.addEventListener("lostpointercapture", () => {
          resetJoystick();
        });
      });
  },
});
